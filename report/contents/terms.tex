% Обозначения и определения --- объекты задачи
\newglossaryentry{агент}{
    name={Агент},
    description={автономный исполнитель, перемещающийся по графу дорожной сети}
}
\newglossaryentry{маршрут}{
    name={Маршрут (индивидуальный план)},
    description={последовательность вершин и действий агента, ведущая от его исходной к целевой вершине}
}
\newglossaryentry{расписание}{
    name={Расписание},
    description={неубывающая последовательность плановых времён прохождения вершин маршрута}
}
\newglossaryentry{конфликт}{
    name={Конфликт},
    description={ситуация, в которой согласно индивидуальным планам два или более агентов оказываются одновременно в недопустимой конфигурации}
}
\newglossaryentry{ограничение}{
    name={Ограничение},
    description={запрет агенту находиться в заданной вершине или выполнять заданное действие в указанный момент или интервал времени}
}
\newglossaryentry{небезоп}{
    name={Небезопасный интервал},
    description={максимальный интервал времени, в течение которого начало действия агента приводит к столкновению с действием другого агента}
}
\newglossaryentry{эвристика}{
    name={Эвристика},
    description={функция, оценивающая снизу стоимость оставшегося пути в алгоритмах информированного поиска}
}
\newglossaryentry{makespan}{
    name={Makespan},
    description={момент времени, к которому все агенты достигают своих целей}
}
\newglossaryentry{flowtime}{
    name={Flowtime},
    description={суммарное время, затраченное всеми агентами на достижение целей}
}
\newglossaryentry{srate}{
    name={Success rate},
    description={доля экземпляров задачи, решённых алгоритмом в пределах заданного бюджета времени}
}
